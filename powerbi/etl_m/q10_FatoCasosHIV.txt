let
    Fonte = Csv.Document(File.Contents("C:\Users\Administrator\Desktop\Trabalho Banco de Dados II - Power BI\dados\hiv\hiv_casos_categoria_sexo_2018_2025.csv"), [Delimiter=";", Encoding=65001]),
